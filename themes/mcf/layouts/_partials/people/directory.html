{{- /* People directory: all people pages that belong to the current site (front matter key == site.Params.sitesme, containing "people"), grouped by role as configured in data/people_roles.yaml. "role" $r.role "label" $r.label "sort" $sort) -}} {{- end -}} {{- end -}} {{- $members = sort $members "sort" -}} {{- /* sections */ -}} {{- $sections := slice -}} {{- range $i, $g := $groups -}} {{- $in := slice -}} {{- range $members }}{{ if in $g.roles .role }}{{ $in = $in | append . }}{{ end }}{{ end -}} {{- with $in -}} {{- $sections = $sections | append (dict "title" $g.title "id" (anchorize $g.title) "roles" $g.roles "members" .) -}} {{- end -}} {{- end -}} {{- $rest := slice -}} {{- range $members }}{{ if not (in $order .role) }}{{ $rest = $rest | append . }}{{ end }}{{ end -}} {{- with $rest -}} {{- $t := $cfg.other | default "Members" -}} {{- $sections = $sections | append (dict "title" $t "id" (anchorize $t) "roles" slice "members" .) -}} {{- end -}}
