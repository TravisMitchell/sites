{{- /* Renders one author name of a publication. Input (dict): id - author ID, e.g. "T Mitchell" (required) name - printed name, e.g. "T.J. Mitchell" (defaults to id) highlight - author ID to emphasise (the "current" person); optional page - current page; if no highlight is given, page.Params.short is used Output: for the highlighted person if the person has a page in content/people/ (pub-member: in the current site's "people" group) otherwise */ -}} {{- $id := .id -}} {{- $name := .name | default .id -}} {{- $highlight := .highlight | default "" -}} {{- if not $highlight -}} {{- with .page -}}{{- with .Params.short -}}{{- $highlight = . -}}{{- end -}}{{- end -}} {{- end -}} {{- $person := site.GetPage (printf "/people/%s" ($id | urlize)) -}} {{- $member := false -}} {{- with $person -}} {{- with index .Params (string site.Params.sitesme) -}} {{- $member = in . "people" -}} {{- end -}} {{- end -}} {{- if and $highlight (eq (lower $id) (lower $highlight)) -}} {{ $name }} {{- else if $person -}} {{ $name }} {{- else -}} {{ $name }} {{- end -}}
